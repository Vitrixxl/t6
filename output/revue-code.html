export async function probeApi(): Promise<boolean> {
  try {
    const response = await fetch(`${API_BASE}/health`, {
      signal: AbortSignal.timeout(PROBE_TIMEOUT_MS),   // 2500 ms : retarde le 1er ecran
      credentials: 'same-origin',
      headers: { Accept: 'application/json' },
    });
    const payload = response.ok ? await response.json().catch(() => null) : null;
    availability = typeof payload === 'object' && payload !== null
      && (payload as { status?: unknown }).status === 'ok';
  } catch { availability = false; }
  return availability;
}

export function authGuard(ctx: AppContext) {
  return new Elysia({ name: 'auth-guard' })
    .use(ctx)
    .resolve({ as: 'scoped' }, ({ cookie, repositories, status }) => {
      const token = cookie[SESSION_COOKIE]?.value;
      if (!token) return status(401, { error: 'Authentification requise.' });

      const session = repositories.sessions.findValid(hashToken(String(token)), new Date().toISOString());
      if (!session) { cookie[SESSION_COOKIE]?.remove(); return status(401, { error: 'Session expiree.' }); }

      return { userId: session.user_id };   // toute route qui monte ce plugin
    });                                     // recoit userId deja type string
}

7 · Sécurité & RGPD

Mots de passe — argon2id fourni par Bun

server/src/security/password.ts:8-16
const ARGON2_OPTIONS = { algorithm: 'argon2id', memoryCost: 19_456, timeCost: 2 } as const;
// Parametres OWASP : 19 Mio, t=2, p=1. Memory-hard -> resiste au GPU,
// contrairement au PBKDF2 du mode autonome.
export function hashPassword(password: string): Promise<string> {
  return Bun.password.hash(password.normalize('NFKC'), ARGON2_OPTIONS);
}

L'empreinte est auto-décrite ($argon2id$v=19$m=...,t=...,p=...$) : un durcissement futur reste rétro-compatible (schema.sql:8-11).

Sessions — jeton opaque, jamais un JWT

server/src/security/tokens.ts:10-17 + services/sessions.ts:28-35
export function createSessionToken() { return randomBytes(32).toString('base64url'); } // 256 bits
export function hashToken(token: string) { return createHash('sha256').update(token).digest('hex'); }

cookie[SESSION_COOKIE]?.set({
  value: token,
  httpOnly: true,                  // un XSS ne vole pas la session
  sameSite: 'lax',                 // pas de CSRF inter-site
  secure: config.isProduction,     // HTTPS hors dev local
  path: '/', maxAge: Math.floor(config.sessionTtlMs / 1000),

Côté serveur — l'idempotence

server/src/services/sync.ts:59-77
export function applyOperations(repositories, userId, batch: Operation[]): SyncOutcome {
  for (const operation of batch) {
    // Une operation deja connue est ignoree, pas rejouee : c'est ce qui rend
    // la file du client sure apres une reponse perdue.
    if (repositories.operations.alreadyApplied(userId, operation.id)) { ignored += 1; continue; }
    apply(repositories, userId, operation);
    repositories.operations.record(userId, operation.id, operation.kind, now.toISOString());
    applied += 1;
  }
  repositories.operations.purgeOlderThan(now);   // sinon le journal grossit sans fin
  return { applied, ignored };
}

9 · Routage & cache partagé

Le navigateur n'appelle jamais OSRM. Trois raisons (services/routing/index.ts:3-12) : quota compté par IP (bug B13), cache mutualisé = éco-conception, URL du calculateur = variable de config.

server/src/services/routing/index.ts:40-61
async route(mode, from, to): Promise<RoutingResult | null> {
  const key = cacheKey(mode, from, to);        // coords arrondies a 5 decimales ~ 1 m
  const cached = cache.find(key);
  if (cached && cached.ageMs < config.routeCacheTtlMs) return { ...JSON.parse(cached.payload), source: 'cache' };

  const geometry = await fetchUpstreamRoute(config.osrmBaseUrl, mode, from, to);
  if (!geometry) {
    // Une entree expiree vaut mieux qu'aucune reponse : la voirie n'a pas
    // change. Seul cas ou l'on sert une donnee perimee.
    if (cached) return { ...JSON.parse(cached.payload), source: 'cache' };
    return null;                                // -> 503 explicite cote route
  }
  cache.save(key, mode, JSON.stringify(geometry));
  cache.purgeOlderThan(config.routeCacheTtlMs);
  return { ...geometry, source: 'upstream' };
}

17 · Questions probables — réponses courtes

Pourquoi SQLite ?

Schéma relationnel standard ; la bascule PostgreSQL ne toucherait que repositories/. bun:sqlite est natif : zéro dépendance à compiler. Argumentaire écrit dans server/src/db/index.ts:4-7.

Pourquoi pas de JWT ?

Jeton opaque 256 bits, seule l'empreinte SHA-256 en base → révocable à la déconnexion et une fuite de base ne rejoue rien. Un JWT autoporteur ne se révoque pas.

Pourquoi aucun CORS ?

Même origine : Vite proxifie /api en dev, l'API est servie sous la même origine en prod. Cookie de 1re partie, aucun en-tête CORS à émettre.

Comment garantissez-vous qu'un rejeu ne duplique rien ?

Chaque opération porte un UUID client ; table applied_operations (user_id, id) en clé primaire ; le lot est appliqué dans une db.transaction. Vu deux fois ⇒ ignored++. Test : server/src/__tests__/sync.test.ts.

Que se passe-t-il si OSRM ne répond pas ?

Le service sert l'entrée de cache expirée si elle existe (seul cas de donnée périmée assumé), sinon null ⇒ 503 explicite ⇒ routingStatus = 'unavailable' ⇒ la carte n'affiche rien pour ce segment et l'UI le dit. Jamais de ligne droite inventée (B14).

Le mode autonome n'est-il pas une faille ?

Non : PBKDF2-SHA-256 120 000 itérations démontre F1 mais n'est pas présenté comme une frontière de sécurité. Dès que l'API répond, c'est elle qui authentifie et le navigateur ne détient qu'un cookie httpOnly.

Limites connues ?

Pas de graphe horaire GTFS (fréquences moyennées) ; covoiturage = comparateur, pas de mise en relation ; rate-limit en mémoire (mono-instance) ; dernier écrivain gagnant en édition concurrente ; deux instances MapLibre montées (O1) ; tuiles OSM publiques (O4). Toutes écrites dans le code et dans docs/BUGS.md.

Où commencer une revue en 15 minutes ?

server/src/routes/auth.ts (sécurité) → server/src/services/sync.ts + src/lib/api/outbox.ts (le couple offline/idempotence) → src/lib/planner/index.ts (le métier).
